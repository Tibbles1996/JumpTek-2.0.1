Embed on your page

<!-- 1. Load the widget script -->
<script src="https://tibbles1996.github.io/JumpTek-2.0.1/jumptek-widget.js"></script>

<!-- 2. Reserve a container -->
<div id="jumptek-host"></div>

<!-- 3. Initialize -->
<script>
  JumpTekWidget.init({
    container: '#jumptek-host',
    serverUrl: 'wss://your-server.example.com',
    fps: 15,
    onEvent: function(data) { console.log(data); }
  });
</script>
